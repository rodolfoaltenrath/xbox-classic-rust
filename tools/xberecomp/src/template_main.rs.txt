use obsidian_native as generated;

use xbox_recomp::runtime::{Desfecho, Limites};

/// Orçamento por fatia e número padrão de fatias. O limite existe para que um serviço
/// faltante não vire laço infinito; `--fatias N` o amplia para títulos mais longos.
const BLOCOS_POR_FATIA: u64 = 1_000_000;
const FATIAS_PADRAO: u32 = 64;

#[derive(Debug, PartialEq, Eq)]
struct Opcoes {
    fatias: u32,
    janela: bool,
    quadro: Option<std::path::PathBuf>,
    quadro_apresentado: Option<std::path::PathBuf>,
    salvar: Option<std::path::PathBuf>,
    carregar: Option<std::path::PathBuf>,
    continuo: bool,
    estados: Option<std::path::PathBuf>,
}

/// Valida toda a linha de comando antes de executar o guest.
fn ler_opcoes(args: impl IntoIterator<Item = String>) -> Result<Opcoes, String> {
    let mut args = args.into_iter();
    let mut opcoes = Opcoes { fatias: FATIAS_PADRAO, janela: false, quadro: None, quadro_apresentado: None, salvar: None, carregar: None, continuo: false, estados: None };
    let mut fatias_definidas = false;
    while let Some(arg) = args.next() {
        match arg.as_str() {
            "--fatias" if !fatias_definidas => {
                let valor = args.next().ok_or("--fatias exige um número")?;
                opcoes.fatias = valor.parse::<u32>().ok().filter(|&n| n > 0)
                    .ok_or_else(|| format!("--fatias inválido: {valor:?}"))?;
                fatias_definidas = true;
            }
            "--janela" if !opcoes.janela => opcoes.janela = true,
            "--continuo" if !opcoes.continuo => opcoes.continuo = true,
            "--quadro" if opcoes.quadro.is_none() => {
                let valor = args.next().filter(|s| !s.is_empty() && !s.starts_with("--"))
                    .ok_or("--quadro exige caminho para BGRA bruto")?;
                opcoes.quadro = Some(valor.into());
            }
            "--quadro-apresentado" if opcoes.quadro_apresentado.is_none() => {
                let valor = args.next().filter(|s| !s.is_empty() && !s.starts_with("--"))
                    .ok_or("--quadro-apresentado exige caminho para BGRA após DAC")?;
                opcoes.quadro_apresentado = Some(valor.into());
            }
            "--salvar" | "--carregar" | "--estados" => {
                let campo = match arg.as_str() { "--salvar" => &mut opcoes.salvar, "--carregar" => &mut opcoes.carregar, _ => &mut opcoes.estados };
                if campo.is_some() { return Err(format!("argumento repetido: {arg}")); }
                let valor = args.next().filter(|s| !s.is_empty() && !s.starts_with("--"))
                    .ok_or_else(|| format!("{arg} exige caminho de estado"))?;
                *campo = Some(valor.into());
            }
            _ => return Err(format!("argumento desconhecido ou repetido: {arg}")),
        }
    }
    if opcoes.continuo && !opcoes.janela { return Err("--continuo exige --janela".into()); }
    if opcoes.estados.is_some() && !opcoes.continuo { return Err("--estados exige --continuo".into()); }
    Ok(opcoes)
}

/// Imprime em stderr o que o guest tocou em MMIO sem modelo, em qualquer saída.
///
/// É a lista exata de registradores que o título usa: sem ela, um acesso que não parou a
/// execução ficaria invisível, e um que parou só diria o endereço do primeiro.
fn relatar_mmio(session: &xbox_recomp::runtime::Session) {
    let gamma = session.bus.nv2a_gamma();
    let different = gamma.palette().map(|p| p.iter().enumerate().filter(|(i, rgb)| **rgb != [*i as u8;3]).count());
    eprintln!("DAC gamma: perfil={:?} cursor={} entradas_nao_identidade={:?}", gamma.profile(), gamma.cursor(), different);
    let clears = session.bus.nv2a_pgraph().stats();
    eprintln!("GPU clear: chamadas={} pixels={} primeira_cor={:?}", clears.calls, clears.pixels, clears.first_color);
    let draws = session.bus.nv2a_pgraph().draw_stats();
    eprintln!("GPU raster: chamadas={} triângulos={} fragmentos={}", draws.calls, draws.triangles, draws.pixels);
    for r in session.bus.nv2a_pfifo().methods() {
        eprintln!("GPU método classe=0x{:04X} método=0x{:04X} subcanal={} contagem={} primeiro=0x{:08X} último=0x{:08X} efeito gráfico={}",r.class,r.method,r.subchannel,r.count,r.first,r.last,if session.bus.nv2a_pgraph().models(r.class,r.method) { "modelado (perfil parcial)" } else { "não modelado" });
    }
    let telemetria = session.bus.telemetria();
    if telemetria.total_de_acessos() > 0 {
        eprintln!("\nMMIO sem modelo tocado pelo guest:\n{}", telemetria.relatorio());
    }
}

/// Resume o framebuffer guest, se o título configurou um modo de vídeo.
///
/// Conta pixels diferentes do preenchimento inicial. É evidência objetiva de que o
/// título **desenhou** algo — a diferença entre "executou" e "produziu imagem" — sem
/// depender de janela, de GPU ou de apresentação (entrega 6 do alpha).
fn relatar_framebuffer(session: &xbox_recomp::runtime::Session) {
    let Ok(Some(view)) = session.kernel.framebuffer(&session.bus) else {
        return;
    };
    let mut coloridos = 0u64;
    let mut cores_vertices = [0u64; 3];
    let mut linhas_com_pixel = 0u32;
    let (mut min_x, mut max_x, mut min_y, mut max_y) = (u32::MAX, 0u32, u32::MAX, 0u32);
    let mut y = 0;
    while let Some(linha) = view.row(y) {
        let mut linha_tem = false;
        for (x, px) in linha.chunks_exact(4).enumerate() {
            // B,G,R,A. Preto opaco ou transparente é fundo; qualquer canal de cor é tinta.
            for (lane, channel) in [2usize, 1, 0].into_iter().enumerate() {
                if px[channel] >= 192 && (0..3).all(|i| i == channel || px[i] <= 32) { cores_vertices[lane] += 1; }
            }
            if px[0] != 0 || px[1] != 0 || px[2] != 0 {
                coloridos += 1;
                linha_tem = true;
                let x = x as u32;
                min_x = min_x.min(x);
                max_x = max_x.max(x);
                min_y = min_y.min(y);
                max_y = max_y.max(y);
            }
        }
        linhas_com_pixel += u32::from(linha_tem);
        y += 1;
    }
    eprintln!(
        "\nFramebuffer guest: modo 0x{:X}, saída {}, {} pixels coloridos em {} linhas",
        view.mode().id(),
        if view.enabled() { "habilitada" } else { "desabilitada" },
        coloridos,
        linhas_com_pixel
    );
    eprintln!("  regiões de cor dos vértices (canal >=192, outros <=32): R={} G={} B={}", cores_vertices[0], cores_vertices[1], cores_vertices[2]);
    if coloridos > 0 {
        eprintln!("  região desenhada: x {min_x}..={max_x}, y {min_y}..={max_y}");
        desenhar_ascii(&view, min_x, max_x, min_y, max_y);
    }
}

/// Desenha a região em ASCII, uma célula por bloco de 2×4 pixels. É como se verifica
/// a olho que o framebuffer contém texto legível e não ruído — o mesmo método do logo
/// do XBE na Fase 1 —, sem janela nem GPU.
fn desenhar_ascii(
    view: &xbox_kernel::dispatch::FramebufferView<'_>,
    min_x: u32,
    max_x: u32,
    min_y: u32,
    max_y: u32,
) {
    const LARGURA_MAXIMA: u32 = 160;
    let passo_x = 1u32;
    let passo_y = 2u32;
    eprintln!();
    let mut y = min_y;
    while y <= max_y {
        let mut texto = String::new();
        let mut x = min_x;
        while x <= max_x && (x - min_x) / passo_x < LARGURA_MAXIMA {
            let mut tinta = false;
            for dy in 0..passo_y {
                if let Some(linha) = view.row(y + dy) {
                    let i = (x * 4) as usize;
                    if let Some(px) = linha.get(i..i + 4) {
                        tinta |= px[0] != 0 || px[1] != 0 || px[2] != 0;
                    }
                }
            }
            texto.push(if tinta { '#' } else { ' ' });
            x += passo_x;
        }
        eprintln!("  {}", texto.trim_end());
        y += passo_y;
    }
}

/// Cópia do framebuffer guest como um quadro que o apresentador possui, ou `None` se o
/// título ainda não configurou um modo de vídeo.
///
/// Copia em vez de emprestar: nenhuma referência à RAM guest atravessa para o código da
/// janela (a arena é `Send`, não `Sync`).
fn capturar_quadro(
    session: &xbox_recomp::runtime::Session,
) -> Option<xbox_platform::display::Quadro> {
    let view = session.kernel.framebuffer(&session.bus).ok()??;
    let bytes = view.copiar_visivel()?;
    xbox_platform::display::Quadro::novo(
        xbox_kernel::dispatch::FRAMEBUFFER_WIDTH,
        xbox_kernel::dispatch::FRAMEBUFFER_HEIGHT,
        bytes,
    )
    .ok()
}

/// Mesma cópia usada pela janela e pela exportação do DAC. Sem efeito na sessão.
fn capturar_saida(session: &xbox_recomp::runtime::Session) -> Option<xbox_platform::display::Quadro> {
    let mut quadro = capturar_quadro(session)?;
    if let Some(palette) = session.bus.nv2a_gamma().palette() {
        quadro.aplicar_rampa_rgb(palette);
    }
    Some(quadro)
}

/// Apresenta o quadro atual e processa a janela. `true` = o usuário pediu para fechar.
///
/// Não apresenta antes de existir um modo de vídeo: a janela só é criada quando há o que
/// mostrar, porque o fundo antes do primeiro quadro é indefinido.
fn apresentar(
    session: &xbox_recomp::runtime::Session,
    janela: &mut Option<Box<dyn xbox_platform::display::Apresentador>>,
    abrir: bool,
) -> Result<bool, Box<dyn std::error::Error>> {
    Ok(apresentar_eventos(session, janela, abrir)? == xbox_platform::display::Evento::Fechar)
}

fn apresentar_eventos(
    session: &xbox_recomp::runtime::Session,
    janela: &mut Option<Box<dyn xbox_platform::display::Apresentador>>,
    abrir: bool,
) -> Result<xbox_platform::display::Evento, Box<dyn std::error::Error>> {
    use xbox_platform::display::Evento;
    if !abrir {
        return Ok(Evento::Nenhum);
    }
    if let Some(quadro) = capturar_saida(session) {
        if janela.is_none() {
            *janela = Some(xbox_platform::criar_janela("Obsidian — Xbox", quadro.largura(), quadro.altura())?);
        }
        if let Some(j) = janela.as_mut() { j.apresentar(&quadro)?; }
    }
    if let Some(j) = janela.as_mut() {
        return Ok(j.eventos()?);
    }
    Ok(Evento::Nenhum)
}

fn exportar_quadro(session: &xbox_recomp::runtime::Session, caminho: &std::path::Path, apresentado: bool) -> Result<(), Box<dyn std::error::Error>> {
    use std::io::Write;
    let quadro = if apresentado { capturar_saida(session) } else { capturar_quadro(session) }
        .ok_or("framebuffer indisponível para exportação")?;
    let mut arquivo = std::fs::OpenOptions::new().write(true).create_new(true).open(caminho)?;
    arquivo.write_all(quadro.bytes())?;
    eprintln!("Quadro BGRA {}: {}x{}, {} bytes -> {}", if apresentado { "apresentado" } else { "bruto" }, quadro.largura(), quadro.altura(), quadro.bytes().len(), caminho.display());
    Ok(())
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let opcoes = ler_opcoes(std::env::args().skip(1))?;
    let mut session = if let Some(path) = &opcoes.carregar {
        let session = xbox_recomp::checkpoint::load_file(path, include_bytes!("image.xbe"))?;
        eprintln!("Estado carregado: {} (instruções={}, ciclos={}, EIP=0x{:08X})", path.display(), session.state.retired, session.state.clock.cycles(), session.state.eip);
        session
    } else { xbox_recomp::runtime::Session::load(include_bytes!("image.xbe"))? };
    let mut janela: Option<Box<dyn xbox_platform::display::Apresentador>> = None;

    let limites = Limites {
        blocos_por_fatia: BLOCOS_POR_FATIA,
        fatias: opcoes.fatias,
    };
    let resultado = if opcoes.continuo {
        executar_continuo(&mut session, &mut janela, limites, opcoes.estados)
    } else { executar(&mut session, &mut janela, opcoes.janela, limites) };
    eprintln!("Kernel: {} chamadas atendidas, {} recusadas; instruções={}, EIP=0x{:08X}",session.kernel.atendidas(),session.kernel.recusadas(),session.state.retired,session.state.eip);
    relatar_mmio(&session);
    relatar_framebuffer(&session);
    if let Some(path) = opcoes.salvar {
        if let Err(erro) = xbox_recomp::checkpoint::save_new(&path, &session, include_bytes!("image.xbe")) {
            eprintln!("Falha ao salvar estado: {erro}");
            if resultado.is_ok() { return Err(erro.into()); }
        } else {
            eprintln!("Estado salvo: {} (instruções={}, ciclos={}, EIP=0x{:08X})", path.display(), session.state.retired, session.state.clock.cycles(), session.state.eip);
        }
    }
    for (caminho, apresentado) in [(opcoes.quadro, false), (opcoes.quadro_apresentado, true)] {
        if let Some(caminho) = caminho {
        if let Err(erro) = exportar_quadro(&session, &caminho, apresentado) {
            eprintln!("Falha ao exportar quadro: {erro}");
            // Uma falha de captura não pode esconder HLT, falta de memória ou
            // serviço recusado. Quando o guest teve sucesso, a captura decide o erro.
            if resultado.is_ok() { return Err(erro); }
        }
        }
    }

    // Com janela, o orçamento do `main` gerado acaba em segundos e a imagem sumiria antes
    // de alguém vê-la. Mantém a janela viva, repintando, até o usuário fechar.
    if opcoes.janela && janela.is_some() && !matches!(&resultado, Ok(Desfecho::ParadoPeloHost)) {
        eprintln!("\nExecução encerrada; janela mantida aberta. Feche-a para sair.");
        loop {
            if apresentar(&session, &mut janela, true)? {
                break;
            }
            std::thread::sleep(std::time::Duration::from_millis(30));
        }
    }
    resultado.map(|_| ())
}

fn executar_continuo(
    session: &mut xbox_recomp::runtime::Session,
    janela: &mut Option<Box<dyn xbox_platform::display::Apresentador>>,
    limites: Limites,
    estados: Option<std::path::PathBuf>,
) -> Result<Desfecho, Box<dyn std::error::Error>> {
    use xbox_platform::display::Evento;
    let mut control = xbox_recomp::control::Control::new(estados);
    let mut pending = Evento::Nenhum;
    eprintln!("Modo contínuo: Espaço pausa/retoma; F5 salva; F9 restaura; X encerra.");
    loop {
        let event = if pending == Evento::Nenhum { apresentar_eventos(session, janela, true)? }
            else { std::mem::replace(&mut pending, Evento::Nenhum) };
        match event {
            Evento::Fechar => {
                eprintln!("Janela fechada pelo usuário; execução interrompida pelo host.");
                return Ok(Desfecho::ParadoPeloHost);
            }
            Evento::PausarRetomar => {
                control.toggle_pause();
                eprintln!("Sessão {}: instruções={}, ciclos={}, EIP=0x{:08X}", if control.paused() { "pausada" } else { "retomada" }, session.state.retired, session.state.clock.cycles(), session.state.eip);
            }
            Evento::SalvarEstado => match control.save(session, include_bytes!("image.xbe")) {
                Ok(path) => eprintln!("Estado salvo: {} (instruções={}, ciclos={}, EIP=0x{:08X})", path.display(), session.state.retired, session.state.clock.cycles(), session.state.eip),
                Err(error) => eprintln!("Falha ao salvar estado por F5: {error}"),
            },
            Evento::RestaurarEstado => match control.restore(session, include_bytes!("image.xbe")) {
                Ok(()) => eprintln!("Estado restaurado: instruções={}, ciclos={}, EIP=0x{:08X}", session.state.retired, session.state.clock.cycles(), session.state.eip),
                Err(error) => eprintln!("Falha ao restaurar estado por F9: {error}"),
            },
            Evento::Nenhum => {},
        }
        if let Some(j) = janela.as_mut() {
            j.definir_titulo(if control.paused() { "Obsidian — PAUSADO | Espaço: retomar | F5: salvar | F9: restaurar" }
                else { "Obsidian — Xbox | Espaço: pausar | F5: salvar | F9: restaurar" })?;
        }
        if control.paused() {
            std::thread::sleep(std::time::Duration::from_millis(16));
            continue;
        }
        let result = control.advance(session, limites, generated::run, |s| {
            pending = apresentar_eventos(s, janela, true).map_err(|e| xbox_recomp::Error::Presentation(e.to_string()))?;
            Ok(pending != Evento::Nenhum)
        }).map_err(|e| e.to_string())?;
        match result {
            Some(Desfecho::Retornou) => return Ok(Desfecho::Retornou),
            Some(Desfecho::ParadoPeloHost) => {}, // evento pendente, tratado antes de avançar
            None => eprintln!("Trecho concluído: instruções={}, ciclos={}, EIP=0x{:08X}", session.state.retired, session.state.clock.cycles(), session.state.eip),
        }
    }
}

fn executar(
    session: &mut xbox_recomp::runtime::Session,
    janela: &mut Option<Box<dyn xbox_platform::display::Apresentador>>,
    abrir_janela: bool,
    limites: Limites,
) -> Result<Desfecho, Box<dyn std::error::Error>> {
    // `main` devolve `Box<dyn Error>` e o Rust imprime esse retorno com `Debug`. Passar
    // pelo texto mantém o diagnóstico legível, em vez do `Debug` derivado da variante.
    let desfecho = session.conduzir(
        limites,
        generated::run,
        |s| {
            apresentar(s, janela, abrir_janela)
                .map_err(|e| xbox_recomp::Error::Presentation(e.to_string()))
        },
    )
    .map_err(|e| e.to_string())?;
    match desfecho {
        Desfecho::Retornou => println!(
            "Execução nativa concluída: EAX={}, instruções={}, chamadas atendidas={}",
            session.state.regs[0], session.state.retired, session.kernel.atendidas()
        ),
        // O usuário fechou a janela: é o host interrompendo um título que não retorna
        // sozinho, não uma falha. Parada controlada, estado intacto.
        Desfecho::ParadoPeloHost => {
            eprintln!("Janela fechada pelo usuário; execução interrompida pelo host.");
        }
    }
    Ok(desfecho)
}
