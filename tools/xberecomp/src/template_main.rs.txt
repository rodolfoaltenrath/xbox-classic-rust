mod generated;

use xbox_recomp::runtime::{Desfecho, Limites};

/// Orçamento por fatia e número padrão de fatias. O limite existe para que um serviço
/// faltante não vire laço infinito; `--fatias N` o amplia para títulos mais longos.
const BLOCOS_POR_FATIA: u64 = 1_000_000;
const FATIAS_PADRAO: u32 = 64;

/// Lê `--fatias N` da linha de comando. Valor ausente ou inválido é erro, não padrão
/// silencioso: um orçamento diferente do pedido mudaria o resultado sem avisar.
fn ler_fatias() -> Result<u32, Box<dyn std::error::Error>> {
    let mut args = std::env::args().skip(1);
    while let Some(arg) = args.next() {
        if arg == "--fatias" {
            let valor = args.next().ok_or("--fatias exige um número")?;
            return match valor.parse::<u32>() {
                Ok(n) if n > 0 => Ok(n),
                _ => Err(format!("--fatias inválido: {valor:?}").into()),
            };
        }
    }
    Ok(FATIAS_PADRAO)
}

/// Imprime em stderr o que o guest tocou em MMIO sem modelo, em qualquer saída.
///
/// É a lista exata de registradores que o título usa: sem ela, um acesso que não parou a
/// execução ficaria invisível, e um que parou só diria o endereço do primeiro.
fn relatar_mmio(session: &xbox_recomp::runtime::Session) {
    let clears = session.bus.nv2a_pgraph().stats();
    eprintln!("GPU clear: chamadas={} pixels={} primeira_cor={:?}", clears.calls, clears.pixels, clears.first_color);
    let draws = session.bus.nv2a_pgraph().draw_stats();
    eprintln!("GPU raster: chamadas={} triângulos={} fragmentos={}", draws.calls, draws.triangles, draws.pixels);
    for r in session.bus.nv2a_pfifo().methods() {
        eprintln!("GPU método classe=0x{:04X} método=0x{:04X} subcanal={} contagem={} primeiro=0x{:08X} último=0x{:08X} efeito gráfico={}",r.class,r.method,r.subchannel,r.count,r.first,r.last,if session.bus.nv2a_pgraph().models(r.class,r.method) { "modelado (perfil parcial)" } else { "não modelado" });
    }
    let telemetria = session.bus.telemetria();
    if telemetria.total_de_acessos() > 0 {
        eprintln!("\nMMIO sem modelo tocado pelo guest:\n{}", telemetria.relatorio());
    }
}

/// Resume o framebuffer guest, se o título configurou um modo de vídeo.
///
/// Conta pixels diferentes do preenchimento inicial. É evidência objetiva de que o
/// título **desenhou** algo — a diferença entre "executou" e "produziu imagem" — sem
/// depender de janela, de GPU ou de apresentação (entrega 6 do alpha).
fn relatar_framebuffer(session: &xbox_recomp::runtime::Session) {
    let Ok(Some(view)) = session.kernel.framebuffer(&session.bus) else {
        return;
    };
    let mut coloridos = 0u64;
    let mut cores_vertices = [0u64; 3];
    let mut linhas_com_pixel = 0u32;
    let (mut min_x, mut max_x, mut min_y, mut max_y) = (u32::MAX, 0u32, u32::MAX, 0u32);
    let mut y = 0;
    while let Some(linha) = view.row(y) {
        let mut linha_tem = false;
        for (x, px) in linha.chunks_exact(4).enumerate() {
            // B,G,R,A. Preto opaco ou transparente é fundo; qualquer canal de cor é tinta.
            for (lane, channel) in [2usize, 1, 0].into_iter().enumerate() {
                if px[channel] >= 192 && (0..3).all(|i| i == channel || px[i] <= 32) { cores_vertices[lane] += 1; }
            }
            if px[0] != 0 || px[1] != 0 || px[2] != 0 {
                coloridos += 1;
                linha_tem = true;
                let x = x as u32;
                min_x = min_x.min(x);
                max_x = max_x.max(x);
                min_y = min_y.min(y);
                max_y = max_y.max(y);
            }
        }
        linhas_com_pixel += u32::from(linha_tem);
        y += 1;
    }
    eprintln!(
        "\nFramebuffer guest: modo 0x{:X}, saída {}, {} pixels coloridos em {} linhas",
        view.mode().id(),
        if view.enabled() { "habilitada" } else { "desabilitada" },
        coloridos,
        linhas_com_pixel
    );
    eprintln!("  regiões de cor dos vértices (canal >=192, outros <=32): R={} G={} B={}", cores_vertices[0], cores_vertices[1], cores_vertices[2]);
    if coloridos > 0 {
        eprintln!("  região desenhada: x {min_x}..={max_x}, y {min_y}..={max_y}");
        desenhar_ascii(&view, min_x, max_x, min_y, max_y);
    }
}

/// Desenha a região em ASCII, uma célula por bloco de 2×4 pixels. É como se verifica
/// a olho que o framebuffer contém texto legível e não ruído — o mesmo método do logo
/// do XBE na Fase 1 —, sem janela nem GPU.
fn desenhar_ascii(
    view: &xbox_kernel::dispatch::FramebufferView<'_>,
    min_x: u32,
    max_x: u32,
    min_y: u32,
    max_y: u32,
) {
    const LARGURA_MAXIMA: u32 = 160;
    let passo_x = 1u32;
    let passo_y = 2u32;
    eprintln!();
    let mut y = min_y;
    while y <= max_y {
        let mut texto = String::new();
        let mut x = min_x;
        while x <= max_x && (x - min_x) / passo_x < LARGURA_MAXIMA {
            let mut tinta = false;
            for dy in 0..passo_y {
                if let Some(linha) = view.row(y + dy) {
                    let i = (x * 4) as usize;
                    if let Some(px) = linha.get(i..i + 4) {
                        tinta |= px[0] != 0 || px[1] != 0 || px[2] != 0;
                    }
                }
            }
            texto.push(if tinta { '#' } else { ' ' });
            x += passo_x;
        }
        eprintln!("  {}", texto.trim_end());
        y += passo_y;
    }
}

/// Cópia do framebuffer guest como um quadro que o apresentador possui, ou `None` se o
/// título ainda não configurou um modo de vídeo.
///
/// Copia em vez de emprestar: nenhuma referência à RAM guest atravessa para o código da
/// janela (a arena é `Send`, não `Sync`).
fn capturar_quadro(
    session: &xbox_recomp::runtime::Session,
) -> Option<xbox_platform::display::Quadro> {
    let view = session.kernel.framebuffer(&session.bus).ok()??;
    let bytes = view.copiar_visivel()?;
    xbox_platform::display::Quadro::novo(
        xbox_kernel::dispatch::FRAMEBUFFER_WIDTH,
        xbox_kernel::dispatch::FRAMEBUFFER_HEIGHT,
        bytes,
    )
    .ok()
}

/// Apresenta o quadro atual e processa a janela. `true` = o usuário pediu para fechar.
///
/// Não apresenta antes de existir um modo de vídeo: a janela só é criada quando há o que
/// mostrar, porque o fundo antes do primeiro quadro é indefinido.
fn apresentar(
    session: &xbox_recomp::runtime::Session,
    janela: &mut Option<Box<dyn xbox_platform::display::Apresentador>>,
    abrir: bool,
) -> Result<bool, Box<dyn std::error::Error>> {
    if !abrir {
        return Ok(false);
    }
    let Some(quadro) = capturar_quadro(session) else {
        return Ok(false);
    };
    if janela.is_none() {
        *janela = Some(xbox_platform::criar_janela(
            "Obsidian — Xbox",
            quadro.largura(),
            quadro.altura(),
        )?);
    }
    if let Some(j) = janela.as_mut() {
        j.apresentar(&quadro)?;
        return Ok(j.eventos()? == xbox_platform::display::Evento::Fechar);
    }
    Ok(false)
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let abrir_janela = std::env::args().skip(1).any(|a| a == "--janela");
    let mut session = xbox_recomp::runtime::Session::load(include_bytes!("image.xbe"))?;
    let mut janela: Option<Box<dyn xbox_platform::display::Apresentador>> = None;

    let limites = Limites {
        blocos_por_fatia: BLOCOS_POR_FATIA,
        fatias: ler_fatias()?,
    };
    let resultado = executar(&mut session, &mut janela, abrir_janela, limites);
    eprintln!("Kernel: {} chamadas atendidas, {} recusadas; instruções={}, EIP=0x{:08X}",session.kernel.atendidas(),session.kernel.recusadas(),session.state.retired,session.state.eip);
    relatar_mmio(&session);
    relatar_framebuffer(&session);

    // Com janela, o orçamento do `main` gerado acaba em segundos e a imagem sumiria antes
    // de alguém vê-la. Mantém a janela viva, repintando, até o usuário fechar.
    if abrir_janela && janela.is_some() {
        eprintln!("\nExecução encerrada; janela mantida aberta. Feche-a para sair.");
        loop {
            if apresentar(&session, &mut janela, true)? {
                break;
            }
            std::thread::sleep(std::time::Duration::from_millis(30));
        }
    }
    resultado
}

fn executar(
    session: &mut xbox_recomp::runtime::Session,
    janela: &mut Option<Box<dyn xbox_platform::display::Apresentador>>,
    abrir_janela: bool,
    limites: Limites,
) -> Result<(), Box<dyn std::error::Error>> {
    // `main` devolve `Box<dyn Error>` e o Rust imprime esse retorno com `Debug`. Passar
    // pelo texto mantém o diagnóstico legível, em vez do `Debug` derivado da variante.
    let desfecho = session.conduzir(
        limites,
        generated::run,
        |s| {
            apresentar(s, janela, abrir_janela)
                .map_err(|e| xbox_recomp::Error::Presentation(e.to_string()))
        },
    )
    .map_err(|e| e.to_string())?;
    match desfecho {
        Desfecho::Retornou => println!(
            "Execução nativa concluída: EAX={}, instruções={}, chamadas atendidas={}",
            session.state.regs[0], session.state.retired, session.kernel.atendidas()
        ),
        // O usuário fechou a janela: é o host interrompendo um título que não retorna
        // sozinho, não uma falha. Parada controlada, estado intacto.
        Desfecho::ParadoPeloHost => {
            eprintln!("Janela fechada pelo usuário; execução interrompida pelo host.");
        }
    }
    Ok(())
}
